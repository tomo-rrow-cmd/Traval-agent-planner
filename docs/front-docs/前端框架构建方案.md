### 1.5 Axios HTTP 实例（`src/api/http.ts`）

核心要点：

- **基础 URL**：`/api/v1`
- **超时**：30 秒
- **请求拦截器**：
  - 从 `localStorage` 读取 `access_token` → 注入 `Authorization: Bearer <token>`
  - 若没有 Token 但有匿名会话 ID → 注入 `X-Session-Id` 头
- **响应拦截器**：
  - `response.data.code !== 0` → 抛出 `ApiError(code, message, request_id)`
  - HTTP 401 → 尝试 `tryRefreshToken()`（POST `/auth/tokens`），成功重试原请求，失败清除登录态并跳转首页
- **ApiError 类**：`class ApiError extends Error { code: number; requestId: string }`

关键代码：

```typescript
import axios from 'axios'
import type { AxiosInstance, InternalAxiosRequestConfig } from 'axios'

const BASE_URL = '/api/v1'

const http: AxiosInstance = axios.create({
  baseURL: BASE_URL,
  timeout: 30000,
  headers: { 'Content-Type': 'application/json; charset=utf-8' },
})

http.interceptors.request.use((config: InternalAxiosRequestConfig) => {
  const token = localStorage.getItem('access_token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  const sessionId = localStorage.getItem('anonymous_session_id')
  if (sessionId && !token) {
    config.headers['X-Session-Id'] = sessionId
  }
  return config
})

http.interceptors.response.use(
  (response) => {
    const { data } = response
    if (data.code !== 0) {
      return Promise.reject(new ApiError(data.code, data.message, data.request_id))
    }
    return response
  },
  async (error) => {
    if (error.response?.status === 401) {
      const refreshed = await tryRefreshToken()
      if (refreshed) {
        error.config.headers.Authorization = `Bearer ${localStorage.getItem('access_token')}`
        return http(error.config)
      }
      localStorage.removeItem('access_token')
      localStorage.removeItem('refresh_token')
      window.location.href = '/login'
    }
    return Promise.reject(error)
  }
)

async function tryRefreshToken(): Promise<boolean> {
  const refreshToken = localStorage.getItem('refresh_token')
  if (!refreshToken) return false
  try {
    const res = await axios.post(`${BASE_URL}/auth/tokens`, { refresh_token: refreshToken })
    const { access_token, refresh_token } = res.data.data
    localStorage.setItem('access_token', access_token)
    localStorage.setItem('refresh_token', refresh_token)
    return true
  } catch { return false }
}

export class ApiError extends Error {
  code: number
  requestId: string
  constructor(code: number, message: string, requestId: string) {
    super(message); this.name = 'ApiError'; this.code = code; this.requestId = requestId
  }
}

export default http
```

**SSE 连接实现方案（关键）**：

后端 `POST /chat/sessions/{id}/messages` 返回 `StreamingResponse`（纯 SSE 文本流），**不能使用标准的 `EventSource`**（因为 EventSource 只支持 GET 请求）。

**推荐方案：`fetch` + `ReadableStream.getReader()`**

```typescript
async function handleSSEStream(sessionId: string, content: string) {
  const token = localStorage.getItem('access_token')
  const sessionId_header = localStorage.getItem('anonymous_session_id')
  
  const response = await fetch(`/api/v1/chat/sessions/${sessionId}/messages`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { 'Authorization': `Bearer ${token}` } : { 'X-Session-Id': sessionId_header })
    },
    body: JSON.stringify({ content }),
  })

  const reader = response.body!.getReader()
  const decoder = new TextDecoder()
  let buffer = ''
  let currentEvent = ''

  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    const lines = buffer.split('\n')
    buffer = lines.pop() || ''

    for (const line of lines) {
      if (line.startsWith('event: ')) {
        currentEvent = line.slice(7).trim()
      } else if (line.startsWith('data: ')) {
        const data = JSON.parse(line.slice(6))
        dispatchSSEEvent(currentEvent, data)
      }
    }
  }
}

// 事件分发
function dispatchSSEEvent(event: string, data: any) {
  switch (event) {
    case 'step_start':
      // 更新进度阶段标签
      break
    case 'message':
      // 追加 AI 消息到 messages
      break
    case 'progress':
      // 更新进度百分比（同步到 planStore）
      break
    case 'step_end':
      // 标记阶段完成
      break
    case 'error':
      // 显示错误 Toast
      break
    case 'done':
      // 关闭流
      // 如有 summary → 调用 planStore.mapSSESummaryToPlan()
      // 如有 plan_id → 可导航到方案详情页
      break
  }
}
```

### 2.7 错误处理体系

**三层错误处理架构**：

| 层级 | 机制 | 职责 |
|------|------|------|
| 1. Axios 拦截器 | 响应拦截器 | code≠0 → ApiError；401 → 自动 Token 刷新 |
| 2. Store/组件 | try/catch | 根据 ApiError.code 分类处理（后端错误码） |
| 3. 全局 | ErrorBoundary | 捕获渲染异常，友好降级 |

**后端错误码分类处理**：

```typescript
try {
  await someApi()
} catch (e) {
  if (e instanceof ApiError) {
    switch (e.code) {
      case 20001: // 未登录/Token 过期
        uiStore.showError('请重新登录')
        authStore.clearSession()
        break
      case 10003: // 资源不存在
        uiStore.showError('请求的资源不存在')
        break
      case 40006: // 不支持的导出格式
        uiStore.showWarning('暂不支持的导出格式')
        break
      case 60001: case 60003: // 外部服务/LLM 错误
        uiStore.showWarning('AI 服务暂时不可用，请稍后重试')
        break
      default:
        uiStore.showError(e.message || '服务异常')
    }
  } else {
    uiStore.showError('网络异常，请检查网络连接')
  }
}
```
