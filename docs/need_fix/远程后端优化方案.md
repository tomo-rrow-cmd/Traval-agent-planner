```python
async def list_by_user(self, user_id: str = "", page: int = 1, page_size: int = 20) -> list[Conversation]:
    query = (
        select(Conversation)
        .where(Conversation.status != "deleted")
        .order_by(Conversation.pinned_at.desc().nullslast(), Conversation.updated_at.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
    )
    if user_id:
        query = query.where(Conversation.user_id == user_id)
    result = await self.db.execute(query)
    return list(result.scalars().all())
```

```python
            .where(Message.conversation_id.in_([c.id for c in convs]))
```

| `callout` | 提示卡片 | `icon, text, level(info/warning/success)` |
| `traffic_list` | 车次列表 | `trains[]` |
| `list` | 标签清单 | `items[], style(tags/bullets/numbered)` |

### 4.2 后端生成 content_blocks

**新增文件**: `app/agents/content_formatter.py`

```python
"""ContentFormatter — 将 final_plan 拼装为 content_blocks 列表（纯规则，无 LLM 调用）"""

from typing import Any


class ContentFormatter:
    """方案内容块格式化器"""

    def build_blocks(self, final_plan: dict) -> list[dict]:
        blocks = []

        # 1. 头部
        blocks.append({"type": "header", "data": self._build_header(final_plan)})

        # 2. 行程概览文本
        overview_text = self._build_overview_text(final_plan)
        if overview_text:
            blocks.append({"type": "text", "data": {"content": overview_text}})

        # 3. 每日行程卡片
        for day in final_plan.get("daily_itinerary", []):
            blocks.append({"type": "day_card", "data": self._build_day_card(day)})

        blocks.append({"type": "divider", "data": {}})

        # 4. 深度详情 callout
        for detail in final_plan.get("attraction_details", []):
            blocks.append({"type": "callout", "data": {
                "icon": "landmark",
                "level": "info",
                "title": detail.get("name", ""),
                "text": detail.get("description", "")
            }})

        # 5. 大交通
        bt = final_plan.get("big_traffic", [])
        if bt:
            blocks.append({"type": "traffic_list", "data": {"trains": bt[:8]}})
            blocks.append({"type": "divider", "data": {}})

        # 6. 市内交通
        traffic = final_plan.get("traffic", {})
        if traffic.get("daily_traffic"):
            blocks.append({"type": "text", "data": {
                "content": self._build_traffic_text(traffic)
            }})
            blocks.append({"type": "divider", "data": {}})

        # 7. 预算图
        budget = final_plan.get("budget", {})
        if budget.get("details"):
            blocks.append({"type": "budget_chart", "data": self._build_budget_data(budget)})
            blocks.append({"type": "divider", "data": {}})

        # 8. 超预算警告
        for w in budget.get("warnings", []):
            blocks.append({"type": "callout", "data": {
                "icon": "warning",
                "level": "warning",
                "text": w
            }})

        # 9. 出行贴士
        tips = final_plan.get("tips", [])
        if tips:
            blocks.append({"type": "text", "data": {
                "content": "## 出行贴士\n\n" + "\n".join(f"- {t}" for t in tips)
            }})

        # 10. 行李清单
        packing = final_plan.get("packing_list", [])
        if packing:
            blocks.append({"type": "list", "data": {
                "items": packing,
                "style": "tags"
            }})

        # 11. 地图（最后）
        markers = self._build_map_markers(final_plan)
        if markers:
            blocks.append({"type": "map", "data": {
                "markers": markers,
                "route_path": markers
            }})

        return blocks

    def _build_header(self, p: dict) -> dict:
        budget = p.get("budget", {})
        return {
            "title": p.get("trip_name", ""),
            "destination": p.get("destination", ""),
            "origin": p.get("origin", ""),
            "days": p.get("days", 0),
            "estimated_cost": budget.get("actual_total", 0),
            "total_budget": budget.get("total_budget", 0),
        }

    def _build_overview_text(self, p: dict) -> str:
        dest = p.get("destination", "")
        origin = p.get("origin", "")
        days = p.get("days", 0)
        summary = p.get("summary", "")
        lines = [f"**{origin + ' → ' if origin else ''}{dest}** · **{days}天**"]
        if summary:
            lines.append(f"> {summary}")
        return "\n\n".join(lines)

    def _build_day_card(self, day: dict) -> dict:
        return {
            "day": day.get("day", 0),
            "date": day.get("date", ""),
            "label": day.get("date_label", ""),
            "weather_summary": day.get("weather_summary", ""),
            "day_cost": day.get("day_cost", 0),
            "day_summary": day.get("day_summary", ""),
            "activities": [
                {
                    "title": a.get("name", ""),
                    "description": a.get("reason", ""),
                    "cost": a.get("price", 0),
                    "duration_minutes": int(a.get("duration_hours", 0) * 60) if a.get("duration_hours") else None,
                    "location": {"lat": a.get("latitude"), "lng": a.get("longitude")} if a.get("latitude") and a.get("longitude") else None,
                    "type": "attraction",
                    "tags": [],
                }
                for a in day.get("attractions", [])
            ],
            "accommodation": {
                "hotel_name": day.get("hotel", {}).get("name", ""),
                "price": day.get("hotel", {}).get("price_per_night", 0),
                "address": day.get("hotel", {}).get("address", ""),
                "rating": day.get("hotel", {}).get("rating", 0),
            } if day.get("hotel") else None,
        }

    def _build_traffic_text(self, traffic: dict) -> str:
        daily = traffic.get("daily_traffic", [])
        lines = ["## 市内交通"]
        for d in daily:
            dd = d.get("day", "?")
            lines.append(f"\n### 第{dd}天")
            for seg in d.get("segments", []):
                best = (seg.get("options", [{}])[0] if seg.get("options") else seg)
                lines.append(
                    f"- {seg.get('from','')} → {seg.get('to','')}："
                    f"**{best.get('mode_label', best.get('mode', ''))}** "
                    f"{best.get('time_min', '')}分钟 ¥{best.get('cost', 0)}"
                )
            lines.append(
                f"_当日交通：{d.get('total_time_min', 0)}分钟，共¥{d.get('total_cost', 0)}_"
            )
        return "\n".join(lines)

    def _build_budget_data(self, budget: dict) -> dict:
        details = {d["category"]: d["spent"] for d in budget.get("details", [])}
        return {
            "total": budget.get("total_budget", 0),
            "breakdown": {
                "transport": details.get("交通", 0),
                "accommodation": details.get("住宿", 0),
                "food": details.get("餐饮", 0),
                "attractions": details.get("门票", 0),
                "souvenirs": details.get("购物", 0),
                "contingency": 0,
            },
        }

    def _build_map_markers(self, p: dict) -> list:
        markers = []
        for day in p.get("daily_itinerary", []):
            for a in day.get("attractions", []):
                if a.get("latitude") and a.get("longitude"):
                    markers.append({
                        "position": [a["longitude"], a["latitude"]],
                        "title": a.get("name", ""),
                        "type": "attraction",
                    })
        return markers
```
